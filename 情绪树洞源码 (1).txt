  // ---------- 七日图表 ----------
  function moodColor(mood) {
    if (mood.indexOf('平静') >= 0) return '#c79a4a';
    if (mood.indexOf('低落') >= 0) return '#8b8bd0';
    if (mood.indexOf('焦虑') >= 0) return '#e1914f';
    if (mood.indexOf('愤怒') >= 0) return '#c85f74';
    if (mood.indexOf('期待') >= 0) return '#5eb18b';
    return '#b7adb3';
  }
  function renderChart() {
    var list = myRecords(); var box = $('chartBox');
    // 柱状图
    var days = []; var now = new Date(); now.setHours(0, 0, 0, 0);
    for (var i = 6; i >= 0; i--) { var d = new Date(now.getTime() - i * 86400000); var key = d.getFullYear() + '-' + d.getMonth() + '-' + d.getDate(); days.push({ key: key, label: (d.getMonth() + 1) + '/' + d.getDate(), n: 0 }); }
    var idx = {}; for (var a = 0; a < days.length; a++) idx[days[a].key] = a;
    for (var b = 0; b < list.length; b++) { var dd = new Date(list[b].ts); var k = dd.getFullYear() + '-' + dd.getMonth() + '-' + dd.getDate(); if (idx[k] != null) days[idx[k]].n++; }
    var max = 1; for (var c = 0; c < days.length; c++) max = Math.max(max, days[c].n);
    var W = 640, H = 230, padL = 30, padB = 26, padT = 14, padR = 10;
    var bw = (W - padL - padR) / days.length; var inner = bw * 0.5;
    var htmlSvg = '<svg viewBox="0 0 ' + W + ' ' + H + '" width="100%" role="img" aria-label="过去七天每天记录条数柱状图"><g font-size="11" fill="currentColor" font-family="system-ui">';
    for (var x = 0; x < days.length; x++) {
      var cx = padL + bw * x + bw / 2; var h = Math.max(3, (days[x].n / max) * (H - padT - padB)); var y = H - padB - h;
      htmlSvg += '<line x1="' + cx + '" y1="' + (H - padB) + '" x2="' + cx + '" y2="' + y + '" stroke="var(--line)" stroke-width="' + inner + '" rx="4"/>';
      htmlSvg += '<line x1="' + cx + '" y1="' + y + '" x2="' + cx + '" y2="' + (H - padB) + '" stroke="var(--accent)" stroke-width="' + inner + '" rx="4"/>';
      htmlSvg += '<text x="' + cx + '" y="' + (y - 7) + '" text-anchor="middle" fill="var(--ink-soft)">' + days[x].n + '</text>';
      htmlSvg += '<text x="' + cx + '" y="' + (H - 8) + '" text-anchor="middle">' + days[x].label + '</text>';
    }
    htmlSvg += '</g></svg>';
    box.innerHTML = htmlSvg;

    // 情绪分布
    var counts = {}; for (var e = 0; e < list.length; e++) { var mk = list[e].mood; counts[mk] = (counts[mk] || 0) + 1; }
    var keysAll = ['😌 平静安然', '😔 低落难过', '😰 焦虑紧绷', '😡 压抑愤怒', '✨ 温柔期待'];
    var keys = keysAll.filter(function (k) { return counts[k]; });
    // 表格
    var tb = $('moodTableBody'); var tbHtml = '';
    for (var f = 0; f < keysAll.length; f++) { var kk = keysAll[f]; var n = counts[kk] || 0; tbHtml += '<tr><td>' + esc(kk) + '</td><td>' + n + '</td></tr>'; }
    if (!list.length) tbHtml = '<tr><td colspan="2">暂无数据</td></tr>';
    tb.innerHTML = tbHtml;
    // 环形图
    var svg = $('donutSvg'); var total = list.length;
    $('donutTotal').textContent = total;
    var ld = $('moodLegend');
    if (!total) { svg.innerHTML = ''; ld.innerHTML = '<span class="li"><span class="dot empty"></span>暂无数据</span>'; return; }
    var colors = keys.map(moodColor);
    var R = 56, cx = 75, cy = 75, C = 2 * Math.PI * R; var offset = 0; var segs = '';
    for (var g = 0; g < keys.length; g++) {
      var frac = counts[keys[g]] / total; var len = frac * C;
      segs += '<circle cx="' + cx + '" cy="' + cy + '" r="' + R + '" fill="none" stroke="' + colors[g] + '" stroke-width="26" stroke-dasharray="' + len.toFixed(2) + ' ' + (C - len).toFixed(2) + '" stroke-dashoffset="' + (-offset).toFixed(2) + '" stroke-linecap="butt"/>';
      offset += len;
    }
    svg.innerHTML = segs;
    var lHtml = ''; for (var h = 0; h < keys.length; h++) lHtml += '<span class="li"><span class="dot" style="background:' + colors[h] + '"></span>' + esc(keys[h]) + '<span class="n">' + counts[keys[h]] + '</span></span>';
    ld.innerHTML = lHtml;
  }

  // ---------- 正念呼吸 ----------
  var breath = { inited: false, root: null, dot: null, stage: null, clock: null, voiceBtnRef: null, running: false, paused: false, remain: 300, phase: 0, tick: null, voice: false };
  function breathInit(root) {
    if (breath.inited && breath.root === root) return;
    var prefix = root === 'modal' ? 'M' : '';
    breath.root = root;
    breath.dot = $('breathDot' + prefix); breath.stage = $('breathStage' + prefix); breath.clock = $('breathClock' + prefix);
    breath.voiceBtnRef = $('voiceBtn' + prefix);
    breath.running = false; breath.paused = false; breath.remain = 300; breath.stage.textContent = '准备开始'; breath.clock.textContent = '05:00';
    if (breath.dot) { breath.dot.style.transform = 'scale(1)'; }
    breath.inited = true;
  }
  function fmtClock(s) { var m = Math.floor(s / 60); var ss = s % 60; return (m < 10 ? '0' : '') + m + ':' + (ss < 10 ? '0' : '') + ss; }
  function say(text) { if (!breath.voice) return; try { if (window.speechSynthesis) { window.speechSynthesis.cancel(); var u = new SpeechSynthesisUtterance(text); u.lang = 'zh-CN'; u.rate = 0.9; window.speechSynthesis.speak(u); } } catch (e) {} }
  function tickBreath() {
    if (!breath.running || breath.paused) return;
    breath.remain = Math.max(0, breath.remain - 1);
    breath.clock.textContent = fmtClock(breath.remain);
    breath.phase = (breath.phase + 1) % 8;
    var mapP = ['吸气', '呼吸', '呼吸', '屏息', '呼气', '呼吸', '呼吸', '屏息'];
    var st = mapP[breath.phase];
    breath.stage.textContent = st;
    if (breath.dot) {
      var s = (st === '吸气') ? 'scale(1.6)' : (st === '呼气' ? 'scale(0.62)' : 'scale(1.12)');
      breath.dot.style.transform = s;
    }
    if (breath.phase % 2 === 0) say(st);
    if (breath.remain <= 0) { stopBreath(); breath.stage.textContent = '练习完成，放松～'; }
  }
  function startBreath(root) {
    breathInit(root);
    if (!breath.running) {
      breath.running = true; breath.paused = false; breath.remain = 300;
      breath.clock.textContent = fmtClock(breath.remain);
      $('breathPause' + (root === 'modal' ? 'M' : '')).textContent = '⏸ 暂停';
      if (breath.tick) clearInterval(breath.tick);
      breath.tick = setInterval(tickBreath, 1000);
      nextPhase(); say('开始正念呼吸');
      toast('开始正念呼吸～');
    }
  }
  function nextPhase() {
    if (!breath.running || breath.paused) return;
    breath.phase = 0; breath.stage.textContent = '吸气'; if (breath.dot) breath.dot.style.transform = 'scale(1.6)'; say('吸气');
  }
  function stopBreath() { breath.running = false; breath.paused = false; if (breath.tick) clearInterval(breath.tick); breath.tick = null; try { if (window.speechSynthesis) window.speechSynthesis.cancel(); } catch (e) {} }
  function togglePause(root) {
    if (!breath.running) return;
    breath.paused = !breath.paused;
    $('breathPause' + (root === 'modal' ? 'M' : '')).textContent = breath.paused ? '▶ 继续' : '⏸ 暂停';
  }
  function bindBreath(prefix) {
    $('voiceBtn' + prefix).addEventListener('click', function () {
      breath.voice = !breath.voice;
      this.textContent = breath.voice ? '🔇 语音引导' : '🔊 语音引导';
      if (breath.voice) say('语音引导已开启');
    });
    $('breathPause' + prefix).addEventListener('click', function () { togglePause(prefix === 'M' ? 'modal' : 'panel'); });
    $('breathReset' + prefix).addEventListener('click', function () { stopBreath(); breathInit(prefix === 'M' ? 'modal' : 'panel'); toast('已重置'); });
  }
  bindBreath(''); bindBreath('M');
